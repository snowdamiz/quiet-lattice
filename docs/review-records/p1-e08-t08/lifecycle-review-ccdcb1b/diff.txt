+    def receive(self, signum, _frame):

+def drain(owned, record):
+    """Bounded cleanup, independent of resource sampling or result hashing."""
+    deadline = time.monotonic() + TERM_GRACE + KILL_GRACE
+    kill_at = time.monotonic() + TERM_GRACE
+    verified = not record['observationUncertain']
+    while time.monotonic() < deadline:
+        try:
+            owned.observe()
+        except BaseException as error:
+            verified = False
+            record['errors'].append('cleanup observation: ' + repr(error))
+        try:
+            if owned.all_exited():
+                # Observed group/descendants are gone; only now reap the leader.
+                record['exit'] = owned.child.wait(timeout=.2)
+                record['directChildReaped'] = True
+                record['remainingProcesses'] = []
+                return verified
+        except BaseException as error:
+            verified = False
+            record['errors'].append('cleanup ownership: ' + repr(error))
+        try:
+            owned.signal_all(signal.SIGKILL if time.monotonic() >= kill_at else signal.SIGTERM)
+        except BaseException as error:
+            verified = False
+            record['errors'].append('cleanup signal: ' + repr(error))
+        time.sleep(POLL)
+    # Keep the ledger charged even if an OS/process observation cannot settle.
+    try:
+        if not owned.waiter_lost:
+            record['exit'] = owned.child.wait(timeout=.2)
+            record['directChildReaped'] = True
+    except subprocess.TimeoutExpired:
+        pass
+    record['remainingProcesses'] = owned.last
+    return False

+def publish_terminal(directory, record, cancellation):
+    """Cleanup is already terminal. Completion linearizes at this masked decision.
+
+    Signals observed before this decision cancel. Signals during the short local
+    persistence transaction are deferred; a successfully sealed completion wins.
+    A crash or write/fsync failure leaves the durable admission unresolved.
+    """
+    previous = signal.pthread_sigmask(signal.SIG_BLOCK, CANCEL_SIGNALS)
+    try:
+        pending = signal.sigpending() & CANCEL_SIGNALS
+        record['cancelSignals'] = sorted(set(cancellation.signals) | {int(s) for s in pending})
+        if record['cleanup'] != 'verified':
+            record['outcome'] = 'uncertain'
+        elif record['cancelSignals']:
+            record['outcome'] = 'cancelled'
+        elif record['exit'] != 0 or record['errors'] or record['stopReason']:
+            record['outcome'] = 'failed'
+        else:
+            record['outcome'] = 'success'
+        receipt = REPORTS / (record['stage'] + '.json')
+        write_json(receipt, record)
+        if record['cleanup'] == 'verified':
+            settle(directory, receipt, record['runId'])
+            cancellation.committed = True
+        return record['outcome'] == 'success'
+    finally:
+        signal.pthread_sigmask(signal.SIG_SETMASK, previous)

+"""Durable admission/settlement for this task's one local process lane."""
+import contextlib
+import fcntl
+import hashlib
+import json
+import os
+from pathlib import Path
+import uuid
+
+
+def digest(path):
+    return hashlib.sha256(path.read_bytes()).hexdigest()
+
+
+def sync_directory(path):
+    fd = os.open(path, os.O_RDONLY)
+    try:
+        os.fsync(fd)
+    finally:
+        os.close(fd)
+
+
+def write_json(path, value, replace=False):
+    """Persist bytes before publishing the name; never replace another run."""
+    temp = path.with_name(path.name + '.' + uuid.uuid4().hex + '.pending')
+    with temp.open('x') as f:
+        json.dump(value, f, indent=2)
+        f.write('\n')
+        f.flush()
+        os.fsync(f.fileno())
+    if replace:
+        os.replace(temp, path)
+        sync_directory(path.parent)
+    else:
+        os.link(temp, path)
+        # A visible seal is insufficient if its directory sync failed. Retain
+        # this durable pending name until all publication operations succeed.
+        sync_directory(path.parent)
+        temp.unlink()
+        # No fallible operation follows removal. A crash may resurrect the
+        # pending name; that conservatively closes admission for human review.
+
+
+def read(path):
+    return json.loads(path.read_text())
+
+
+@contextlib.contextmanager
+def lane_lock(reports):
+    reports.mkdir(parents=True, exist_ok=True)
+    with (reports / 'lane.lock').open('a') as lock:
+        fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
+        yield
+
+
+def settled_runs(reports, *, promotion=False):
+    """Missing/corrupt/partially published ownership never means an empty lane.
+
+    Legacy receipts are frozen inputs, not new lifecycle proof. New failures may
+    release admission after proved cleanup, but may never pass finalization.
+    """
+    if list(reports.glob('*.pending')):
+        raise RuntimeError('terminal publication remains uncertain')
+    legacy = read(Path(__file__).with_name('legacy-stages.json'))['stages']
+    for files in legacy.values():
+        for name, expected in files.items():
+            if digest(reports / name) != expected:
+                raise RuntimeError('historical evidence changed: ' + name)
+    registered = {}
+    for directory in sorted((reports / 'runs').glob('*')):
+        if not directory.is_dir():
+            raise RuntimeError('unknown ownership entry: ' + str(directory))
+        if list(directory.glob('*.pending')) or (directory / 'publication-failure.json').exists():
+            raise RuntimeError('ownership publication remains uncertain: ' + directory.name)
+        try:
+            admitted = read(directory / 'admitted.json')
+            stage = admitted['stage']
+            if stage != directory.name or admitted['schema'] != 2:
+                raise ValueError('admission identity')
+            receipt = reports / (stage + '.json')
+            terminal = read(receipt)
+            seal = read(directory / 'settled.json')
+            if (terminal['runId'] != admitted['runId'] or terminal['stage'] != stage
+                    or any(terminal.get(key) != value for key, value in admitted.items())
+                    or terminal['schema'] != 2 or seal['runId'] != admitted['runId']
+                    or seal['admissionSHA256'] != digest(directory / 'admitted.json')
+                    or seal['terminalSHA256'] != digest(receipt)
+                    or terminal['cleanup'] != 'verified' or terminal['remainingProcesses']
+                    or not terminal['directChildReaped']):
+                raise ValueError('unsettled or mismatched terminal evidence')
+            if promotion and (terminal['outcome'] != 'success' or terminal['exit'] != 0
+                              or terminal['errors'] or terminal['cancelSignals'] or terminal['stopReason']):
+                raise ValueError('failed/cancelled run cannot promote')
+            registered[stage] = terminal
+        except (OSError, ValueError, KeyError, TypeError) as error:
+            raise RuntimeError(f'ownership unresolved: {directory.name}: {error}') from error
+    for log in reports.glob('*.log'):
+        if log.stem not in legacy and log.stem not in registered:
+            raise RuntimeError('unregistered or unfinished stage log: ' + log.name)
+    for path in reports.glob('*.json'):
+        value = read(path)
+        if isinstance(value, dict) and 'stage' in value and 'command' in value:
+            if path.stem not in legacy and path.stem not in registered:
+                raise RuntimeError('unregistered terminal record: ' + path.name)
+    return registered
+
+
+def admit(reports, name, command):
+    settled_runs(reports)
+    if (reports / (name + '.log')).exists() or (reports / (name + '.json')).exists():
+        raise FileExistsError('stage name already used: ' + name)
+    runs = reports / 'runs'
+    runs.mkdir(exist_ok=True)
+    sync_directory(reports)
+    directory = runs / name
+    directory.mkdir()  # Even a crash before admitted.json leaves admission closed.
+    sync_directory(runs)
+    admission = dict(schema=2, stage=name, runId=uuid.uuid4().hex,
+                     supervisorPID=os.getpid(), command=command)
+    write_json(directory / 'admitted.json', admission)
+    return directory, admission
+
+
+def settle(directory, receipt, run_id):
+    write_json(directory / 'settled.json', dict(
+        runId=run_id, admissionSHA256=digest(directory / 'admitted.json'),
+        terminalSHA256=digest(receipt)))

+"""macOS process ownership: unreaped direct child and watched descendants.
+
+No PID/command-name adoption after restart. An abandoned admission stays closed.
+"""
+import ctypes
+import errno
+import os
+import select
+import signal
+import subprocess
+
+
+class BSDInfo(ctypes.Structure):
+    # sys/proc_info.h, PROC_PIDTBSDINFO. The start pair survives exec and separates
+    # PID reuse; kqueue watches the process instance, not a later reused PID.
+    _fields_ = [(name, ctypes.c_uint32) for name in (
+        'flags', 'status', 'xstatus', 'pid', 'ppid', 'uid', 'gid', 'ruid',
+        'rgid', 'svuid', 'svgid', 'reserved')]
+    _fields_ += [('comm', ctypes.c_char * 16), ('name', ctypes.c_char * 32)]
+    _fields_ += [(name, ctypes.c_uint32) for name in (
+        'nfiles', 'pgid', 'jobc', 'tdev', 'tpgid', 'nice')]
+    _fields_ += [('start_sec', ctypes.c_uint64), ('start_usec', ctypes.c_uint64)]
+
+
+LIBPROC = ctypes.CDLL('/usr/lib/libproc.dylib', use_errno=True)
+LIBPROC.proc_pidinfo.argtypes = [ctypes.c_int, ctypes.c_int, ctypes.c_uint64, ctypes.c_void_p, ctypes.c_int]
+LIBPROC.proc_pidinfo.restype = ctypes.c_int
+
+
+def identity(pid):
+    info = BSDInfo()
+    count = LIBPROC.proc_pidinfo(pid, 3, 0, ctypes.byref(info), ctypes.sizeof(info))
+    if count == 0 and ctypes.get_errno() in (errno.ESRCH, errno.ENOENT):
+        return None
+    if count != ctypes.sizeof(info) or info.pid != pid:
+        raise OSError(ctypes.get_errno(), 'cannot identify owned process', pid)
+    return (int(info.start_sec), int(info.start_usec))
+
+
+def snapshot():
+    result = subprocess.run(['ps', '-axo', 'pid=,ppid=,pgid=,rss=,comm='],
+                            capture_output=True, text=True, check=True, timeout=2)
+    rows = []
+    for line in result.stdout.splitlines():
+        a = line.split(None, 4)
+        if len(a) != 5:
+            raise RuntimeError('incomplete process observation')
+        rows.append(dict(pid=int(a[0]), ppid=int(a[1]), pgid=int(a[2]),
+                         rss=int(a[3]) * 1024, command=a[4]))
+    return rows
+

+class ProcessWatch:
+    def __init__(self, pid):
+        self.pid = pid
+        self.birth = identity(pid)
+        self.queue = select.kqueue()
+        self.ended = self.birth is None
+        try:
+            if not self.ended:
+                self.queue.control([select.kevent(pid, filter=select.KQ_FILTER_PROC,
+                    flags=select.KQ_EV_ADD | select.KQ_EV_ONESHOT,
+                    fflags=select.KQ_NOTE_EXIT)], 0, 0)
+                if identity(pid) != self.birth:
+                    raise RuntimeError('process changed while acquiring watch')
+        except ProcessLookupError:
+            self.ended = True
+        except BaseException:
+            self.queue.close()
+            raise
+
+    def exited(self):
+        if not self.ended:
+            for event in self.queue.control([], 1, 0):
+                if (event.ident != self.pid or event.filter != select.KQ_FILTER_PROC
+                        or event.flags & select.KQ_EV_ERROR or not event.fflags & select.KQ_NOTE_EXIT):
+                    raise RuntimeError('unexpected process-watch event; exit is unproved')
+                self.ended = True
+        return self.ended
+
+    def send(self, sig):
+        if self.exited():
+            return
+        if identity(self.pid) != self.birth:
+            raise RuntimeError('owned process identity changed; refusing signal')
+        try:
+            os.kill(self.pid, sig)
+        except ProcessLookupError:
+            self.ended = True
+
+    def evidence(self):
+        return dict(pid=self.pid, birth=self.birth, exitObserved=self.exited())
+
+
+class OwnedProcesses:
+    def __init__(self, child):
+        self.child = child  # Retain the actual Popen through every error path.
+        self.birth = identity(child.pid)
+        self.waiter_lost = False
+        self.watches = {}
+        self.retired = []
+        self.last = []
+
+    def leader_exited(self):
+        # WNOWAIT retains the actual child and reserves its PID/process-group
+        # identity until every known descendant is terminal. poll() would reap.
+        try:
+            return os.waitid(os.P_PID, self.child.pid, os.WEXITED | os.WNOHANG | os.WNOWAIT) is not None
+        except ChildProcessError:
+            self.waiter_lost = True
+            raise
+
+    def observe(self):
+        self.leader_exited()  # Prove the group anchor is still our unreaped child.
+        rows = snapshot()
+        ids = {self.child.pid}
+        ids.update(pid for pid, watch in self.watches.items() if not watch.exited())
+        ids.update(p['pid'] for p in rows if p['pgid'] == self.child.pid)
+        while True:
+            added = {p['pid'] for p in rows if p['ppid'] in ids} - ids
+            if not added:
+                break
+            ids.update(added)
+        owned = [p for p in rows if p['pid'] in ids]
+        for row in owned:
+            pid = row['pid']
+            if pid in self.watches and self.watches[pid].exited():
+                watch = self.watches[pid]
+                if identity(pid) not in (None, watch.birth):
+                    self.retired.append(watch.evidence())
+                    watch.queue.close()
+                    del self.watches[pid]
+            if pid != self.child.pid and pid not in self.watches:
+                self.watches[pid] = ProcessWatch(pid)
+        self.last = owned
+        return owned
+
+    def signal_all(self, sig):
+        # The direct child is deliberately not reaped, even after leader exit.
+        # Its reserved PID anchors this group. Detached known children also have
+        # individual instance watches, so group termination alone is insufficient.
+        errors = []
+        try:
+            # Darwin clears BSD start fields on a zombie. waitid(WNOWAIT), not
+            # those cleared fields, proves our still-unreaped direct child.
+            leader_done = self.leader_exited()
+            if not leader_done and identity(self.child.pid) != self.birth:
+                raise RuntimeError('direct-child ownership lost; refusing group signal')
+            grouped = any(row['pgid'] == self.child.pid and row['pid'] != self.child.pid
+                          and not self.watches[row['pid']].exited() for row in self.last)
+            try:
+                if not leader_done or grouped:
+                    os.killpg(self.child.pid, sig)
+            except ProcessLookupError:
+                pass
+            except PermissionError:
+                # Darwin returns EPERM for a group containing only our zombie
+                # leader. Its waitable exit plus no live group watch settles it.
+                if grouped or not self.leader_exited():
+                    raise
+        except Exception as error:
+            errors.append(repr(error))
+        for watch in self.watches.values():
+            try:
+                watch.send(sig)
+            except Exception as error:
+                errors.append(repr(error))
+        if errors:
+            raise RuntimeError('; '.join(errors))
+
+    def all_exited(self):
+        return self.leader_exited() and all(w.exited() for w in self.watches.values())
+
+    def evidence(self):
+        return dict(directChild=dict(pid=self.child.pid, birth=self.birth, waiterLost=self.waiter_lost),
+                    knownDescendants=self.retired + [w.evidence() for w in self.watches.values()])
+
+    def close(self):
+        for watch in self.watches.values():
+            watch.queue.close()
