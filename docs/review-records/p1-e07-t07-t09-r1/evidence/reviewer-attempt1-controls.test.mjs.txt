function oracle(marker, args) {
  const literals = [...spec.matchAll(/`([^`]+)`/gs)].map(m=>m[1]);
  const source = literals.find(s=>s.startsWith(marker));
  if (!source) throw Error('Production predicate not found');
  const substituted = new Function(...Object.keys(args), 'return `'+source+'`;')(...Object.values(args));
  return new Function('window','return ('+substituted+');');
}
